\subsection{Notations}
\begin{frame}
    \frametitle{Rappel et Notation}
    \begin{itemize}
        \item On utiliseras les symboles: \(\Pi\) pour la projection,  \(\sigma\) pour la selection et \(\Join\) pour la jointure entre deux tables
        \item A chaque étape de l'arbre, \(n\) feras référence au nombre de ligne encore présente dans la table
        \item Les arbres d'éxécution se lisent du bas vers le haut, la dernière projection représentant la toute dernière opération efféctué
        \item Le jeux de données utilisé:
              \begin{itemize}
                  \item \(3\) Tables
                  \item \(100 000\) lignes par table
                  \item \(10 000\) requête lancé par paramètre
                  \item \(100\) requêtes différentes et très variées
                  \item \(1\) point vaut une moyenne de \(100\) requête
              \end{itemize}
    \end{itemize}
\end{frame}

\subsection{Méthodes de jointure}
\begin{frame}
    \frametitle{Les différents types de jointures possibles}
    \begin{itemize}
        \item Produit Cartésien: génère tous les couples possibles et teste la condition de jointure \(\rightarrow \mathcal{O}(n^2)\)\\
        \item Jointure Triée: pré-trie les deux tables et lit les deux tables \(\rightarrow \mathcal{O}(2n\log(n)+2n)\)
        \item Jointure par dictionnaire: crée un dictionnaire et recherche les correspondances \(\rightarrow \mathcal{O}(2n)\)
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Comment choisir le meilleur plan rapidement?}
    \begin{itemize}
        \item Choisir les optimisations avant de lancer la requête
        \item Utiliser l'historique des plans générés
        \item Donner un coût à chaque opération, énumérer tous les plans et prendre celui avec le plus faible coût
    \end{itemize}

\subsection{Critique}
\begin{frame}
    \frametitle{Les raisons de la lenteur de notre modèle}
    \begin{itemize}
        \item De nombreuses techniques n'ont pas été traitées, notamment les plus récentes (\(\sim 20\) ans)
        \item Les failles de notre code. \(45\) minutes \(\to\) \(22\) secondes
        \item Les optimisations matérielles:Appel aux fonctions internes au processeur, le multi-threading
        \item La compression des données. Exemple: \textit{C-Store} \(4\) manière de compresser les données
        \item L'indexation des données facilitent les jointures
    \end{itemize}
